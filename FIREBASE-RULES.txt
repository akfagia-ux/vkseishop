═══════════════════════════════════════════════════════════════
                        FIRESTORE RULES
═══════════════════════════════════════════════════════════════

ГДЕ: Firestore Database → Rules

ВСТАВЬ ЭТО:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Функция проверки роли
    function isOwner() {
      return request.auth != null && 
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'владелец';
    }
    
    function isAdmin() {
      return request.auth != null && 
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role in ['владелец', 'администратор'];
    }
    
    function isModerator() {
      return request.auth != null && 
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role in ['владелец', 'администратор', 'модератор'];
    }
    
    // Пользователи
    match /users/{userId} {
      allow read: if true;
      allow create: if request.auth != null;
      allow update: if request.auth != null && 
                      (request.auth.uid == userId || isAdmin());
      allow delete: if isOwner();
    }
    
    // Общий чат
    match /chat/{messageId} {
      allow read: if true;
      allow create: if request.auth != null || request.resource.data.userId == 'system';
      allow update: if isModerator() || resource.data.userId == request.auth.uid;
      allow delete: if isModerator();
    }
    
    // Закрепленные сообщения
    match /pinnedMessages/{messageId} {
      allow read: if true;
      allow write: if isOwner();
    }
    
    // Личные чаты
    match /chats/{chatId} {
      allow read: if request.auth != null && 
                    request.auth.uid in resource.data.participants;
      allow create: if request.auth != null || 
                      request.resource.data.participants != null;
      allow update: if request.auth != null && 
                      request.auth.uid in resource.data.participants;
      allow delete: if isOwner();
      
      // Сообщения в личных чатах
      match /messages/{messageId} {
        allow read: if request.auth != null && 
                      request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants;
        allow create: if request.auth != null || 
                        request.resource.data.userId == 'system';
        allow delete: if isOwner();
      }
    }
    
    // Заказы
    match /orders/{orderId} {
      allow read: if true;
      allow create: if true;
      allow update: if true;
      allow delete: if isOwner();
    }
    
    // Отзывы
    match /reviews/{reviewId} {
      allow read: if true;
      allow create: if request.auth != null;
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }
    
    // Тикеты поддержки
    match /tickets/{ticketId} {
      allow read: if request.auth != null && 
                    (resource.data.userId == request.auth.uid || isModerator());
      allow create: if request.auth != null;
      allow update: if request.auth != null && 
                      (resource.data.userId == request.auth.uid || isModerator());
      allow delete: if isOwner();
      
      // Сообщения в тикетах
      match /messages/{messageId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null;
        allow delete: if isOwner();
      }
    }
    
    // Скидки
    match /discounts/{discountId} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }
    
    // Telegram пользователи
    match /telegram_users/{userId} {
      allow read: if true;
      allow write: if true;
    }
  }
}

═══════════════════════════════════════════════════════════════
                        STORAGE RULES
═══════════════════════════════════════════════════════════════

ГДЕ: Storage → Rules

ВСТАВЬ ЭТО:

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    
    // Аватары пользователей
    match /avatars/{userId}/{fileName} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == userId;
      allow delete: if request.auth != null && request.auth.uid == userId;
    }
    
    // Баннеры пользователей
    match /banners/{userId}/{fileName} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == userId;
      allow delete: if request.auth != null && request.auth.uid == userId;
    }
    
    // Файлы в тикетах поддержки
    match /tickets/{ticketId}/{fileName} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }
    
    // Общие файлы
    match /uploads/{fileName} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
